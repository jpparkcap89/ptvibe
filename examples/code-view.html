const PUBLIC_KEYS_ = ['appName','subtitle','trainingDate','trainingPlace','target','capacity','registrationOpen','privacyController','privacyPurpose','privacyRetention','privacyContact','privacyVersion','notice','contentOverrides','configRevision'];
function db_() {
  const id = PropertiesService.getScriptProperties().getProperty('DB_ID');

function settings_() {
  const rows = db_().getSheetByName('Settings').getDataRange().getDisplayValues();
  return Object.fromEntries(rows.slice(1).filter(r => r[0]).map(r => [r[0],r[1]]));
}
function getPublicData() {
  const s = settings_();

}

function submitRegistration(p) {
  if (!p || typeof p !== 'object') throw new Error('신청 내용을 확인해 주세요.');
  const name = String(p.name || '').trim();
  const school = String(p.school || '').trim();
  const position = String(p.position || '');
  const phone = String(p.phone || '').replace(/[^0-9]/g,'');
  const requestId = String(p.requestId || '');
  if (name.length < 2 || name.length > 30 || school.length < 2 || school.length > 80) throw new Error('성명과 소속을 확인해 주세요.');
  if (!['교사','수석교사','교감','교장','기타'].includes(position)) throw new Error('직급을 선택해 주세요.');
  if (!/^0\d{8,10}$/.test(phone)) throw new Error('연락처를 확인해 주세요.');
  if (p.consent !== true) throw new Error('개인정보 수집·이용 동의가 필요합니다.');
  if (!/^[a-zA-Z0-9-]{16,80}$/.test(requestId)) throw new Error('신청 화면을 다시 열어 주세요.');
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = db_().getSheetByName('Registrations');
    const rows = sh.getDataRange().getDisplayValues().slice(1);
    if (rows.some(r => r[9] === requestId)) return {ok:true,message:'신청이 접수되었습니다.'};
    const s = settings_();
    if (s.registrationOpen !== 'TRUE') throw new Error('현재 신청을 받지 않습니다.');
    if (!s.privacyController || !s.privacyRetention || !s.privacyContact || !s.privacyPurpose || !s.privacyVersion) throw new Error('운영자의 개인정보 안내 설정이 필요합니다.');
    if (String(p.privacyVersion) !== s.privacyVersion) throw new Error('개인정보 안내가 변경되었습니다. 새로고침 후 확인해 주세요.');
    if (rows.some(r => r[5] === phone)) throw new Error('이미 신청된 연락처입니다. 변경이 필요하면 운영자에게 문의해 주세요.');
    const capacity = Number(s.capacity);
    if (!Number.isInteger(capacity) || capacity < 1) throw new Error('정원 설정이 필요합니다.');
    if (rows.length >= capacity) throw new Error('신청 정원이 마감되었습니다.');
    // 문자열을 수식으로 해석하지 않도록 처리합니다.
    const safe = v => /^[=+\-@]/.test(v) ? "'"+v : v;
    sh.getRange(sh.getLastRow()+1,1,1,11).setNumberFormat('@').setValues([[
      Utilities.getUuid(),Utilities.formatDate(new Date(),'Asia/Seoul','yyyy-MM-dd HH:mm:ss'),safe(name),safe(school),position,phone,'동의',s.privacyVersion,'접수',requestId,
      JSON.stringify({controller:s.privacyController,purpose:s.privacyPurpose,items:'성명, 소속, 직급, 연락처',retention:s.privacyRetention,contact:s.privacyContact})
    ]]);
    SpreadsheetApp.flush();
    return {ok:true,message:'신청이 접수되었습니다.'};
  } finally { lock.releaseLock(); }
}

function fingerprint_(value) {
  return Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, value));

function loginAdmin(password) {
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    const cache = CacheService.getScriptCache();
    const failures = Number(cache.get('training_login_failures') || 0);
    if (failures >= 10) throw new Error('로그인 시도가 많습니다. 5분 후 다시 시도해 주세요.');
    const expected = adminPassword_();
    if (!expected) throw new Error('Settings의 adminPassword 값을 설정해 주세요.');
    if (typeof password !== 'string' || password.length > 200 || fingerprint_(password) !== fingerprint_(expected)) {
      cache.put('training_login_failures',String(failures+1),300);
      throw new Error('관리자 비밀번호를 확인해 주세요.');
    }
    cache.remove('training_login_failures');
    const token = Utilities.getUuid() + Utilities.getUuid();
    cache.put('training_admin_'+token, fingerprint_(expected),1800);
    return {token:token};
  } finally { lock.releaseLock(); }
}
function requireAdmin_(token) {
  if (typeof token !== 'string' || !/^[a-f0-9-]{72}$/i.test(token)) throw new Error('관리자 로그인이 필요합니다.');
  const saved = CacheService.getScriptCache().get('training_admin_'+token);
  const password = adminPassword_();
  if (!saved || !password || saved !== fingerprint_(password)) throw new Error('관리자 로그인이 만료되었습니다. 다시 로그인해 주세요.');
}

function saveAdminData(token,payload) {
  requireAdmin_(token);
  if (!payload || typeof payload !== 'object' || !payload.settings || typeof payload.settings !== 'object') throw new Error('설정 내용을 확인해 주세요.');
  const patch = {};
  PUBLIC_KEYS_.filter(k=>k!=='configRevision').forEach(k=>{
    if (Object.prototype.hasOwnProperty.call(payload.settings,k)) {
      if (typeof payload.settings[k] !== 'string') throw new Error('설정 값은 문자열이어야 합니다.');
      patch[k] = payload.settings[k].trim();
      if (k!=='contentOverrides' && patch[k].length>4000) throw new Error('설정 내용이 너무 깁니다.');
    }
  });
  if (patch.contentOverrides !== undefined) {
    if (patch.contentOverrides.length>45000) throw new Error('페이지 내용이 너무 깁니다.');
    let overrides; try { overrides=JSON.parse(patch.contentOverrides || '{}'); } catch(e) { throw new Error('페이지 내용 형식 오류입니다.'); }
    if (!overrides || Array.isArray(overrides) || typeof overrides!=='object') throw new Error('페이지 내용 형식 오류입니다.');
    Object.entries(overrides).forEach(([key,value])=>{
      if (!/^(home|theory|files|examples|about|guide|apply)-\d+$/.test(key) || typeof value!=='string' || value.length>4000) throw new Error('페이지 내용을 확인해 주세요.');
    });
  }
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    requireAdmin_(token);
    const s = settings_();
    if (String(payload.revision || '0') !== (s.configRevision || '0')) throw new Error('다른 화면에서 내용이 변경되었습니다. 관리자 새로고침 후 다시 수정해 주세요.');
    const merged=Object.assign({},s,patch);
    if (!merged.appName) throw new Error('앱 제목을 입력해 주세요.');
    if (!Number.isInteger(Number(merged.capacity)) || Number(merged.capacity)<1 || Number(merged.capacity)>10000) throw new Error('정원은 1~10000의 정수로 입력해 주세요.');
    if (!['TRUE','FALSE'].includes(merged.registrationOpen)) throw new Error('신청 상태를 확인해 주세요.');

    const policyChanged=['privacyController','privacyPurpose','privacyRetention','privacyContact'].some(k=>merged[k]!==s[k]);

    patch.configRevision=String(Number(s.configRevision || 0)+1);
    const sh=db_().getSheetByName('Settings');
    const rows=sh.getDataRange().getDisplayValues();
    Object.entries(patch).forEach(([key,value])=>{
      const i=rows.findIndex((r,n)=>n>0&&r[0]===key);
